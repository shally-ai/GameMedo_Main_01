// ==========================================
// FIREBASE FIRESTORE SECURITY RULES
// ==========================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Allow public read on content, samples, and hero_media
    match /content/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /samples/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /hero_media/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // Allow public to create bookings, but only admins to read
    match /bookings/{booking} {
      allow create: if true;
      allow read, update, delete: if request.auth != null;
    }

    // Purchases & Referrals: Admin only
    match /purchases/{purchase} {
      allow read, write: if request.auth != null;
    }
    
    match /affiliates/{affiliate} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    
    match /referrals/{referral} {
      allow read, write: if request.auth != null;
    }
  }
}

// ==========================================
// FIREBASE STORAGE SECURITY RULES
// ==========================================

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /{allPaths=**} {
      // Allow public read for uploaded media
      allow read: if true;
      
      // Only authenticated admins can upload
      allow write: if request.auth != null;
    }
  }
}
